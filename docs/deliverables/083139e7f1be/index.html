export type TransferPhase =
  | 'BURN_SUBMITTED'
  | 'BURN_CONFIRMED'
  | 'ATTESTATION_POLLING'
  | 'ATTESTATION_RECEIVED'
  | 'MINT_SUBMITTED'
  | 'MINT_CONFIRMED';

export interface TransferProgressEvent {
  phase: TransferPhase;
  txHash?: string;
  attestationBytes?: string;
  elapsedMs: number;
  metadata?: Record<string, unknown>;
}

export type ProgressCallback = (event: TransferProgressEvent) => void | Promise<void>;

export async function transfer(
  params: TransferParams,
  options?: { onProgress?: ProgressCallback; abortSignal?: AbortSignal }
): Promise<TransferResult> {
  const start = Date.now();
  const emit = (phase: TransferPhase, extra?: Partial<TransferProgressEvent>) => {
    options?.onProgress?.({
      phase,
      elapsedMs: Date.now() - start,
      ...extra
    });
  };

  // 1. Burn
  const burnTx = await submitBurnTransaction(params);
  emit('BURN_SUBMITTED', { txHash: burnTx.hash });
  await waitForConfirmation(burnTx.hash);
  emit('BURN_CONFIRMED', { txHash: burnTx.hash });

  // 2. Attestation
  emit('ATTESTATION_POLLING');
  const attestation = await pollIrisAttestation(burnTx.messageBytes);
  emit('ATTESTATION_RECEIVED', { attestationBytes: attestation });

  // 3. Mint
  const mintTx = await submitMintTransaction(params.destChain, attestation);
  emit('MINT_SUBMITTED', { txHash: mintTx.hash });
  await waitForConfirmation(mintTx.hash);
  emit('MINT_CONFIRMED', { txHash: mintTx.hash });

  return { success: true, burnTxHash: burnTx.hash, mintTxHash: mintTx.hash };
}
